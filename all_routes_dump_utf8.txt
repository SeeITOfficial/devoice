import { NextResponse } from "next/server";
import { prisma } from "@/lib/prisma";
import { validateAdminApiRequest } from "@/lib/adminAuth";
import { validateMediaUrl } from "@/lib/urlValidator";

export async function POST(request: Request) {
  if (!await validateAdminApiRequest(request)) {
    return NextResponse.json({ message: "Unauthorized." }, { status: 401 });
  }
  try {
    const body = await request.json();

    const title = typeof body.title === "string" ? body.title.trim() : "";
    const releaseDate = body.releaseDate ? new Date(body.releaseDate) : null;
    const validArtworkUrl = validateMediaUrl(body.artworkUrl, "image");

    if (!title || !validArtworkUrl || !body.releaseDate) {
      return NextResponse.json(
        {
          success: false,
          message: "Title, valid artwork URL from uploads, and release date are required.",
        },
        { status: 400 }
      );
    }

    if (isNaN(releaseDate?.getTime() ?? NaN)) {
      return NextResponse.json(
        {
          success: false,
          message: "Invalid release date format.",
        },
        { status: 400 }
      );
    }

    const album = await prisma.album.create({
      data: {
        title: title,
        artworkUrl: validArtworkUrl,
        releaseDate: releaseDate as Date,
        published: Boolean(body.published),
      },
    });

    return NextResponse.json(
      {
        success: true,
        album,
      },
      { status: 201 }
    );
  } catch (error) {
    console.error("Album creation error:", error);

    return NextResponse.json(
      {
        success: false,
        message: "Unable to create album.",
      },
      { status: 500 }
    );
  }
}

import { validateAdminApiRequest } from "@/lib/adminAuth";

export async function POST(request: Request) {
  if (!await validateAdminApiRequest(request)) {
    return NextResponse.json({ message: "Unauthorized." }, { status: 401 });
  }
  try {
    const body = await request.json();

    const title =
      typeof body.title === "string"
        ? body.title.trim()
        : "";

    const description =
      typeof body.description === "string"
        ? body.description.trim()
        : "";

    const posterUrl = body.posterUrl
        ? validateMediaUrl(body.posterUrl, "image")
        : null;

    if (body.posterUrl && !posterUrl) {
      return NextResponse.json({ success: false, message: "Invalid poster URL (must be R2 hosted)." }, { status: 400 });
    }

    const date =
      typeof body.date === "string"
        ? new Date(body.date)
        : null;

    if (!title || !body.date) {
      return NextResponse.json(
        {
          success: false,
          message: "Title and date are required.",
        },
        { status: 400 }
      );
    }

    if (!date || Number.isNaN(date.getTime())) {
      return NextResponse.json(
        {
          success: false,
          message: "Invalid event date.",
        },
        { status: 400 }
      );
    }

    if (date <= new Date()) {
      return NextResponse.json(
        {
          success: false,
          message: "Event date must be in the future.",
        },
        { status: 400 }
      );
    }

    const event = await prisma.event.create({
      data: {
        title,
        date,
        description: description || null,
        posterUrl: posterUrl || null,
        ticketTypes: {
          create: {
            name: body.ticketName,
            price: Number(body.ticketPrice),
            quantity: body.ticketUnlimited
              ? null
              : Number(body.ticketQuantity),
            unlimited: Boolean(body.ticketUnlimited),
          },
        },
      },
    });

    return NextResponse.json(
      {
        success: true,
        event,

import { NextResponse } from "next/server";
import { prisma } from "@/lib/prisma";
import { validateAdminApiRequest } from "@/lib/adminAuth";
import { validateMediaUrl } from "@/lib/urlValidator";

export async function GET(
  request: Request,
  { params }: { params: Promise<{ id: string }> }
) {
  if (!await validateAdminApiRequest(request)) {
    return NextResponse.json({ message: "Unauthorized." }, { status: 401 });
  }
  try {
    const { id } = await params;

    const event = await prisma.event.findUnique({
      where: {
        id,
      },
      include: {
        ticketTypes: {
          orderBy: {
            price: "asc",
          },
        },
        orders: {
          where: {
            status: "CONFIRMED",
          },
          include: {
            tickets: {
              where: {
                status: { in: ["VALID", "USED"] },
              },
            },
          },
        },
      },
    });

    if (!event) {
      return NextResponse.json(
        { message: "Event not found." },
        { status: 404 }
      );
    }

    // Calculate accounting statistics
    const stats = {
      totalEntered: 0,
      totalRevenue: 0,
      byTicketType: {} as Record<string, { entered: number; revenue: number }>,
      byPaymentMethod: {
        ONLINE: { entered: 0, revenue: 0 },
        WALK_IN: { entered: 0, revenue: 0 },
      },
    };

    // Initialize ticket types
    event.ticketTypes.forEach(tt => {
      stats.byTicketType[tt.name] = { entered: 0, revenue: 0 };
    });

    event.orders.forEach((order) => {
      const isOnline = order.paymentMethod === "MOBILE_MONEY";
      const paymentKey = isOnline ? "ONLINE" : "WALK_IN";

      order.tickets.forEach((ticket) => {
        const type = event.ticketTypes.find(t => t.id === ticket.ticketTypeId);
        if (!type) return;

        stats.totalRevenue += type.price;
        stats.byTicketType[type.name].revenue += type.price;
        stats.byPaymentMethod[paymentKey].revenue += type.price;

        if (ticket.status === "USED") {
          stats.totalEntered += 1;
          stats.byTicketType[type.name].entered += 1;
          stats.byPaymentMethod[paymentKey].entered += 1;
        }
      });
    });

    // Strip orders from the response body to avoid huge payloads
    const { orders, ...eventWithoutOrders } = event;

    return NextResponse.json({ 
      success: true, 
      event: eventWithoutOrders,
      stats 
    });
  } catch (error) {
    console.error("Event fetch error:", error);

    return NextResponse.json(
      { message: "Unable to load event." },
      { status: 500 }
    );
  }
}

export async function PATCH(
  request: Request,
  { params }: { params: Promise<{ id: string }> }
) {
  if (!await validateAdminApiRequest(request)) {
    return NextResponse.json({ message: "Unauthorized." }, { status: 401 });
  }
  try {
    const { id } = await params;
    const body = await request.json();

    const data: {
      title?: string;
      date?: Date;
      description?: string | null;
      posterUrl?: string | null;
      cancelled?: boolean;
    } = {};

    if (body.title !== undefined) {
      if (!body.title.trim()) {
        return NextResponse.json(
          { message: "Event title cannot be empty." },
          { status: 400 }
        );
      }

      data.title = body.title.trim();
    }

    if (body.date !== undefined) {
      const date = new Date(body.date);

      if (Number.isNaN(date.getTime())) {
        return NextResponse.json(
          { message: "Invalid event date." },
          { status: 400 }
        );
      }

      data.date = date;
    }

    if (body.description !== undefined) {
      data.description = body.description || null;
    }

    if (body.posterUrl !== undefined) {
      if (body.posterUrl) {
        const validated = validateMediaUrl(body.posterUrl, "image");
        if (!validated) {
           return NextResponse.json({ message: "Invalid poster URL (must be R2 hosted)." }, { status: 400 });
        }
        data.posterUrl = validated;
      } else {
        data.posterUrl = null;
      }
    }

    if (body.cancelled !== undefined) {
      data.cancelled = Boolean(body.cancelled);
    }

    const event = await prisma.$transaction(async (transaction) => {
      if (body.ticketTypes !== undefined) {
        if (!Array.isArray(body.ticketTypes)) {
          throw new Error("Invalid ticket types data.");
        }

        for (const ticket of body.ticketTypes) {
          if (!ticket.name || !ticket.name.trim()) {
            throw new Error("Ticket name cannot be empty.");
          }
          if (Number.isNaN(Number(ticket.price)) || Number(ticket.price) < 0) {
            throw new Error("Ticket price must be a valid non-negative number.");
          }
          if (!ticket.unlimited && (Number.isNaN(Number(ticket.quantity)) || Number(ticket.quantity) < 1)) {
            throw new Error("Ticket quantity must be at least 1 for finite tickets.");
          }
        }

        const existingTicketTypes = await transaction.ticketType.findMany({
          where: { eventId: id },
          include: {
            _count: { select: { tickets: true } }
          }
        });

        const incomingIds = new Set(
          body.ticketTypes
            .map((t: any) => t.id)
            .filter((i: any) => i)
        );

        const toDelete = existingTicketTypes.filter(et => !incomingIds.has(et.id));
        for (const ticket of toDelete) {
          if (ticket._count.tickets > 0) {
            throw new Error(`Cannot delete ticket type '${ticket.name}' because it has issued tickets.`);
          }
          await transaction.ticketType.delete({
            where: { id: ticket.id }
          });
        }

        for (const ticket of body.ticketTypes) {
          const ticketData = {
            name: ticket.name.trim(),
            price: Number(ticket.price),
            quantity: ticket.unlimited ? null : Number(ticket.quantity),
            unlimited: Boolean(ticket.unlimited),
          };

          if (ticket.id) {
            const existing = existingTicketTypes.find(et => et.id === ticket.id);
            if (!existing) {
              throw new Error(`Ticket type ${ticket.id} not found.`);
            }

            await transaction.ticketType.update({
              where: { id: ticket.id },
              data: ticketData,
            });
          } else {
            await transaction.ticketType.create({
              data: {
                ...ticketData,
                eventId: id,
              },
            });
          }
        }
      }

      const updatedEvent = await transaction.event.update({
        where: { id },
        data,
      });

      return transaction.event.findUnique({
        where: { id: updatedEvent.id },
        include: { ticketTypes: true },
      });
    });

    return NextResponse.json({
      success: true,
      event,
    });
  } catch (error) {
    console.error("Event update error:", error);

    const isClientError = error instanceof Error && error.message.includes("Cannot delete");
    
    return NextResponse.json(
      {
        success: false,
        message: error instanceof Error ? error.message : "Unable to update event.",
      },
      { status: isClientError ? 400 : 500 }
    );
  }
}

import { NextResponse } from "next/server";
import { scryptSync, timingSafeEqual } from "crypto";
import { prisma } from "@/lib/prisma";

function verifyPassword(password: string, storedPassword: string) {
  const [salt, storedHash] = storedPassword.split(":");

  if (!salt || !storedHash) {
    return false;
  }

  const hash = scryptSync(password, salt, 64).toString("hex");

  return timingSafeEqual(
    Buffer.from(hash, "hex"),
    Buffer.from(storedHash, "hex")
  );
}

export async function POST(request: Request) {
  try {
    const body = await request.json();

    if (!body.email || !body.password) {
      return NextResponse.json(
        {
          message: "Email and password are required.",
        },
        { status: 400 }
      );
    }

    const user = await prisma.managementUser.findUnique({
      where: {
        email: body.email.toLowerCase(),
      },
    });

    if (!user || !verifyPassword(body.password, user.passwordHash)) {
      return NextResponse.json(
        {
          message: "Invalid email or password.",
        },
        { status: 401 }
      );
    }

    const response = NextResponse.json({
      success: true,
    });

    response.cookies.set("devoice_management", user.id, {
      httpOnly: true,
      sameSite: "lax",
      secure: process.env.NODE_ENV === "production",
      path: "/",
      maxAge: 60 * 60 * 24 * 7,
    });

    return response;
  } catch (error) {
    console.error("Admin login error:", error);

    return NextResponse.json(
      {
        message: "Unable to sign in.",
      },
      { status: 500 }
    );
  }
}

import { NextResponse } from "next/server";
import { prisma } from "@/lib/prisma";
import { validateAdminApiRequest } from "@/lib/adminAuth";

export async function POST(request: Request) {
  try {
    const isAdmin = await validateAdminApiRequest(request);

    if (!isAdmin) {
      return NextResponse.json(
        { message: "Unauthorised." },
        { status: 401 }
      );
    }

    const body = await request.json();
    const code = (body.ticketCode ?? body.qrToken ?? "").trim();

    if (!code) {
      return NextResponse.json(
        { message: "Provide a ticketCode or qrToken." },
        { status: 400 }
      );
    }

    // Look up by ticketCode or qrToken
    const ticket = await prisma.ticket.findFirst({
      where: {
        OR: [{ ticketCode: code }, { qrToken: code }],
      },
      include: {
        ticketType: {
          select: { id: true, name: true, price: true, unlimited: true, quantity: true },
        },
        order: {
          include: {
            event: {
              select: {
                id: true,
                title: true,
                date: true,
                cancelled: true,
              },
            },
          },
        },
      },
    });

    if (!ticket) {
      return NextResponse.json(
        { success: false, result: "NOT_FOUND", message: "Ticket not found." },
        { status: 404 }
      );
    }

    const isWalkIn = ticket.order.paymentMethod === "CASH";

    // Reserved / Unpaid
    if (ticket.status === "RESERVED") {
      if (!isWalkIn) {
        return NextResponse.json({
          success: false,
          result: "RESERVED",
          message: "Payment has not been confirmed for this ticket.",
          ticket: {
            id: ticket.id,
            ticketCode: ticket.ticketCode,
            status: ticket.status,
            ticketType: ticket.ticketType,
            customerName: ticket.order.customerName,
            phone: ticket.order.phone,
            email: ticket.order.email,
            event: ticket.order.event,
          },
        });
      }

      // CASH Walk-in logic: Check inventory and validate atomic
      try {
        const result = await prisma.$transaction(async (tx) => {
          if (!ticket.ticketType.unlimited && ticket.ticketType.quantity !== null) {
            const confirmedCount = await tx.ticket.count({
              where: {
                ticketTypeId: ticket.ticketType.id,
                order: { status: "CONFIRMED" },
              },
            });

            // Make sure we have inventory for the ENTIRE order since we confirm the whole order
            if (confirmedCount + ticket.order.quantity > ticket.ticketType.quantity) {
              throw new Error("SOLD_OUT");
            }
          }

          // Confirm the order
          await tx.ticketOrder.update({
            where: { id: ticket.orderId },
            data: { status: "CONFIRMED" },
          });

          // Mark all tickets in the order as VALID
          await tx.ticket.updateMany({
            where: { orderId: ticket.orderId, status: "RESERVED" },
            data: { status: "VALID" },
          });

          // Mark THIS specific ticket as USED
          await tx.ticket.update({
            where: { id: ticket.id },
            data: { status: "USED" },
          });

          return true;
        });

        return NextResponse.json({
          success: true,
          result: "CHECKED_IN",
          message: "CASH / WALK-IN - ENTRY ACCEPTED",
          ticket: {
            id: ticket.id,
            ticketCode: ticket.ticketCode,
            status: "USED",
            ticketType: ticket.ticketType,
            customerName: ticket.order.customerName,
            phone: ticket.order.phone,
            email: ticket.order.email,
            event: ticket.order.event,
          },
        });
      } catch (err) {
        if (err instanceof Error && err.message === "SOLD_OUT") {
          return NextResponse.json({
            success: false,
            result: "SOLD_OUT",
            message: "Tickets are sold out. Cannot accept this walk-in.",
            ticket: null
          });
        }
        throw err;
      }
    }

    // Already consumed
    if (ticket.status === "USED") {
      return NextResponse.json({
        success: false,
        result: "ALREADY_USED",
        message: "This ticket has already been used.",
        ticket: {
          id: ticket.id,
          ticketCode: ticket.ticketCode,
          status: ticket.status,
          ticketType: ticket.ticketType,
          customerName: ticket.order.customerName,
          phone: ticket.order.phone,
          email: ticket.order.email,
          event: ticket.order.event,
        },
      });
    }

    // Invalidated
    if (ticket.status === "INVALID") {
      return NextResponse.json({
        success: false,
        result: "INVALID",
        message: "This ticket is invalid.",
        ticket: {
          id: ticket.id,
          ticketCode: ticket.ticketCode,
          status: ticket.status,
          ticketType: ticket.ticketType,
          customerName: ticket.order.customerName,
          phone: ticket.order.phone,
          email: ticket.order.email,
          event: ticket.order.event,
        },
      });
    }

    // VALID â€” attempt atomic VALID â†’ USED transition
    const updated = await prisma.ticket.updateMany({
      where: {
        id: ticket.id,
        status: "VALID",
      },
      data: { status: "USED" },
    });

    if (updated.count === 0) {
      return NextResponse.json({
        success: false,
        result: "ALREADY_USED",
        message: "This ticket was just used by another check-in attempt.",
        ticket: {
          id: ticket.id,
          ticketCode: ticket.ticketCode,
          status: "USED",
          ticketType: ticket.ticketType,
          customerName: ticket.order.customerName,
          phone: ticket.order.phone,
          email: ticket.order.email,
          event: ticket.order.event,
        },
      });
    }

    return NextResponse.json({
      success: true,
      result: "CHECKED_IN",
      message: isWalkIn ? "CASH / WALK-IN - ENTRY ACCEPTED" : "ONLINE / PAID - ENTRY ACCEPTED",
      ticket: {
        id: ticket.id,
        ticketCode: ticket.ticketCode,
        status: "USED",
        ticketType: ticket.ticketType,
        customerName: ticket.order.customerName,
        phone: ticket.order.phone,
        email: ticket.order.email,
        event: ticket.order.event,
      },
    });
  } catch (error) {
    console.error("Ticket validation error:", error);
    return NextResponse.json(
      { message: "Unable to validate ticket." },
      { status: 500 }
    );
  }
}

import { PutObjectCommand } from "@aws-sdk/client-s3";
import { getSignedUrl } from "@aws-sdk/s3-request-presigner";
import { r2, R2_BUCKET_NAME } from "@/lib/r2";
import { validateAdminApiRequest } from "@/lib/adminAuth";

const ALLOWED_TYPES = {
  image: [
    "image/jpeg",
    "image/png",
    "image/webp",
    "image/heic",
    "image/heif",
  ],
  audio: [
    "audio/mpeg",
    "audio/mp4",
    "audio/wav",
    "audio/x-wav",
    "audio/aac",
    "audio/ogg",
    "audio/webm",
  ],
  video: [
    "video/mp4",
    "video/webm",
    "video/quicktime",
  ],
} as const;

const MAX_SIZE = {
  image: 15 * 1024 * 1024,
  audio: 100 * 1024 * 1024,
  video: 500 * 1024 * 1024,
};

export async function POST(request: Request) {
  if (!await validateAdminApiRequest(request)) {
    return NextResponse.json({ message: "Unauthorized." }, { status: 401 });
  }
  try {
    const body = await request.json();

    const {
      fileName,
      contentType,
      size,
      mediaType,
    } = body;

    if (!fileName || !contentType || !size || !mediaType) {
      return NextResponse.json(
        {
          success: false,
          message:
            "fileName, contentType, size and mediaType are required.",
        },
        { status: 400 }
      );
    }

    // Validate size is a positive integer
    const sizeNum = typeof size === "number" ? size : NaN;
    if (!Number.isFinite(sizeNum) || sizeNum <= 0) {
      return NextResponse.json(
        { success: false, message: "size must be a positive number." },
        { status: 400 }
      );
    }

    if (!["image", "audio", "video"].includes(mediaType)) {
      return NextResponse.json(
        {
          success: false,
          message: "Invalid media type.",
        },
        { status: 400 }
      );
    }

    const allowedTypes =
      ALLOWED_TYPES[mediaType as keyof typeof ALLOWED_TYPES];

    if (!allowedTypes.includes(contentType as never)) {
      return NextResponse.json(
        {
          success: false,
          message: `Unsupported ${mediaType} format.`,
        },
        { status: 400 }
      );
    }

    const maxSize =
      MAX_SIZE[mediaType as keyof typeof MAX_SIZE];

    if (sizeNum > maxSize) {
      return NextResponse.json(
        {
          success: false,
          message: `File is too large. Maximum size is ${Math.round(
            maxSize / 1024 / 1024
          )}MB.`,
        },
        { status: 400 }
      );
    }

    // Sanitize filename: keep only safe characters, strip leading dots from each segment
    const rawBaseName = String(fileName)
      .replace(/[^a-zA-Z0-9._-]/g, "-")
      .replace(/-+/g, "-")
      .split(".")
      .map((seg) => seg.replace(/^\.+/, "") || "x")
      .join(".");

    const extension = rawBaseName.includes(".")
      ? rawBaseName.split(".").pop()?.toLowerCase() ?? "bin"
      : "bin";

    const safeName = rawBaseName;

    const uniqueName = `${Date.now()}-${crypto.randomUUID()}-${safeName}`;

    const folder =
      mediaType === "image"
        ? "images"
        : mediaType === "audio"
          ? "audio"
          : "videos";

    const key = `${folder}/${uniqueName}`;

    const command = new PutObjectCommand({
      Bucket: R2_BUCKET_NAME,
      Key: key,
      ContentType: contentType,
      ContentLength: sizeNum,
    });

    const uploadUrl = await getSignedUrl(r2, command, {
      expiresIn: 60 * 10,
    });

    const publicUrl =
      `${process.env.R2_PUBLIC_URL}/${key}`;

    return NextResponse.json({
      success: true,
      uploadUrl,
      publicUrl,
      key,
      contentType,
      extension,
    });
  } catch (error) {
    console.error("R2 upload URL error:", error);

    return NextResponse.json(
      {
        success: false,
        message: "Unable to create upload URL.",
      },
      { status: 500 }
    );
  }
}

const GENERIC_RESPONSE =
  "If tickets are associated with this number, a verification code has been sent to the registered email address.";

const OTP_EXPIRY_MINUTES = 10;
const RATE_LIMIT_WINDOW_MINUTES = 15;
const RATE_LIMIT_MAX_REQUESTS = 3;

function hashOtp(otp: string): string {
  return createHash("sha256").update(otp).digest("hex");
}

function generateOtp(): string {
  // 6-digit cryptographically random OTP (000000â€“999999)
  return String(randomInt(0, 1_000_000)).padStart(6, "0");
}

export async function POST(request: Request) {
  try {
    const body = await request.json();
    const phone = (body.phone ?? "").trim();

    if (!phone) {
      return NextResponse.json(
        { success: true, message: GENERIC_RESPONSE },
        { status: 200 }
      );
    }

    // --- Rate limiting: max RATE_LIMIT_MAX_REQUESTS per phone per window ---
    const windowStart = new Date(
      Date.now() - RATE_LIMIT_WINDOW_MINUTES * 60 * 1000
    );

    const recentRequests = await prisma.ticketAccessCode.count({
      where: {
        phone,
        createdAt: { gte: windowStart },
      },
    });

    if (recentRequests >= RATE_LIMIT_MAX_REQUESTS) {
      // Still return generic â€” don't reveal rate limit info to enumeration
      return NextResponse.json(
        { success: true, message: GENERIC_RESPONSE },

    const orders = await prisma.ticketOrder.findMany({
      where: {
        phone,
        status: { in: ["PENDING", "CONFIRMED"] },

import { NextResponse } from "next/server";
import { createHash } from "crypto";
import { cookies } from "next/headers";
import { prisma } from "@/lib/prisma";

/**
 * POST /api/ticket/verify-code
 * Body: { phone: string; code: string }
 *
 * Verifies the OTP submitted by the customer.
 * On success: sets a short-lived HTTP-only signed session cookie.
 * On failure: increments attempts counter; blocks after MAX_ATTEMPTS.
 *
 * Never returns ticket data â€” only a success/failure signal.
 */

const MAX_ATTEMPTS = 5;
const SESSION_MAX_AGE_SECONDS = 60 * 60; // 1 hour

function hashOtp(otp: string): string {
  return createHash("sha256").update(otp).digest("hex");
}

function signSession(phone: string): string {
  const secret = process.env.TICKET_SESSION_SECRET;
  if (!secret) {
    throw new Error(
      "TICKET_SESSION_SECRET environment variable is not configured."
    );
  }
  const payload = { phone, verifiedAt: Date.now() };
  const payloadB64 = Buffer.from(JSON.stringify(payload)).toString("base64url");
  const sig = createHash("sha256")
    .update(`${payloadB64}.${secret}`)
    .digest("hex");
  return `${payloadB64}.${sig}`;
}

export function verifySession(token: string): { phone: string } | null {
  try {
    const secret = process.env.TICKET_SESSION_SECRET;
    if (!secret) return null;

    const [payloadB64, sig] = token.split(".");
    if (!payloadB64 || !sig) return null;

    const expectedSig = createHash("sha256")
      .update(`${payloadB64}.${secret}`)
      .digest("hex");

    if (sig !== expectedSig) return null;

    const payload = JSON.parse(
      Buffer.from(payloadB64, "base64url").toString("utf-8")
    ) as { phone: string; verifiedAt: number };

    const ageMs = Date.now() - payload.verifiedAt;
    if (ageMs > SESSION_MAX_AGE_SECONDS * 1000) return null;

    return { phone: payload.phone };
  } catch {
    return null;
  }
}

export async function POST(request: Request) {
  try {
    const body = await request.json();
    const phone = (body.phone ?? "").trim();
    const code = (body.code ?? "").trim();

    if (!phone || !code || !/^\d{6}$/.test(code)) {
      return NextResponse.json(
        { success: false, message: "Invalid code format." },
        { status: 400 }
      );
    }

    // Find the most recent unexpired, unused code for this phone
    const accessCode = await prisma.ticketAccessCode.findFirst({
      where: {
        phone,
        usedAt: null,
        expiresAt: { gt: new Date() },
      },
      orderBy: { createdAt: "desc" },
    });

    if (!accessCode) {
      return NextResponse.json(
        {
          success: false,
          message: "No valid code found. Please request a new one.",
        },
        { status: 400 }
      );
    }

    // Rate-limit verification attempts
    if (accessCode.attempts >= MAX_ATTEMPTS) {
      return NextResponse.json(
        {
          success: false,
          message:
            "Too many incorrect attempts. Please request a new code.",
        },
        { status: 429 }
      );
    }

    const submittedHash = hashOtp(code);
    const isValid = submittedHash === accessCode.codeHash;

    if (!isValid) {
      // Increment attempts
      await prisma.ticketAccessCode.update({
        where: { id: accessCode.id },
        data: { attempts: accessCode.attempts + 1 },
      });

      const remaining = MAX_ATTEMPTS - (accessCode.attempts + 1);
      return NextResponse.json(
        {
          success: false,
          message:
            remaining > 0
              ? `Incorrect code. ${remaining} attempt${remaining === 1 ? "" : "s"} remaining.`
              : "Too many incorrect attempts. Please request a new code.",
        },
        { status: 400 }
      );
    }

    // Valid â€” mark as used (one-time use)
    await prisma.ticketAccessCode.update({
      where: { id: accessCode.id },
      data: { usedAt: new Date() },
    });

    // Issue signed session cookie
    const token = signSession(phone);
    const cookieStore = await cookies();
    cookieStore.set("ticket_session", token, {
      httpOnly: true,
      secure: process.env.NODE_ENV === "production",
      sameSite: "lax",
      maxAge: SESSION_MAX_AGE_SECONDS,
      path: "/",
    });

    return NextResponse.json({ success: true });
  } catch (error) {
    console.error("[verify-code] error:", error);
    return NextResponse.json(
      { success: false, message: "An error occurred. Please try again." },
      { status: 500 }
    );
  }
}
